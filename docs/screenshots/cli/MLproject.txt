name: stock-sentiment-mlops

# MLproject guarantees identical dev / test / training environments.
# Conda env spec lives in conda.yaml (created in Phase 4).

python_env: python_env.yaml

entry_points:
  ingest:
    command: "python -m ssa_ingestion.pipeline"

  features:
    command: "python -m ssa_features.pipeline"

  train:
    parameters:
      model_type: { type: string, default: logistic_regression }
      data_path: { type: path, default: data/processed/features.parquet }
    command: "python -m ssa_model.train --model-type {model_type} --data-path {data_path}"

  evaluate:
    parameters:
      run_id: { type: string }
    command: "python -m ssa_model.evaluate --run-id {run_id}"

  main:
    command: "python -m ssa_model.train"
